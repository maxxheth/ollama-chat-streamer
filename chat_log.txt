fn fibonacci(n: usize) -> Vec<u64> {
    let mut fib = Vec::new();
    let mut a = 0;
    let mut b = 1;

    for _ in 0..n {
        fib.push(a);
        let next = a + b;
        a = b;
        b = next;
    }

    fib
}
